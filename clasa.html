Operații cu fracții — o implementare orientată pe obiecte

Mai jos folosim o clasă proprie, Fractie, pentru a realiza operații aritmetice de bază (adunare, scădere, înmulțire, împărțire) pe fracții reduse la forma ireductibilă. Exemplul ilustrează principiile OOP într-un context matematic elementar.

Definiția clasei Fractie

class Fractie {
  constructor(n, d){
    if(d === 0) throw new Error('Numitorul nu poate fi 0');
    this.n = n; this.d = d;
    this._reduce();
  }
  static _gcd(a,b){ return b ? Fractie._gcd(b, a % b) : Math.abs(a); }
  _reduce(){
    const g = Fractie._gcd(this.n, this.d);
    this.n /= g; this.d /= g;
    if(this.d < 0){ this.n *= -1; this.d *= -1; }
  }
  toString(){ return this.d === 1 ? String(this.n) : `${this.n}/${this.d}`; }
  add(other){ return new Fractie(this.n*other.d + other.n*this.d, this.d*other.d); }
  sub(other){ return new Fractie(this.n*other.d - other.n*this.d, this.d*other.d); }
  mul(other){ return new Fractie(this.n*other.n, this.d*other.d); }
  div(other){ return new Fractie(this.n*other.d, this.d*other.n); }
}
